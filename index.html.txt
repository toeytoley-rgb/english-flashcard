        /* =======================================================
           2. GEMINI API REAL-TIME DICTIONARY LOOKUP & GENERATOR
           ======================================================= */
        async function callGeminiApiForWord(promptText) {
            const apiKey = process.env.GEMINI_API_KEY || "";
            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${apiKey}`;

            const systemInstruction = `You are a professional English-Thai Dictionary for Primary School kids in Thailand. 
Return ONLY valid JSON matching the schema below. Do not wrap in markdown quotes if possible or return strictly pure JSON.`;

            const schema = {
                type: "OBJECT",
                properties: {
                    word: { type: "STRING" },
                    thai: { type: "STRING" },
                    phonetic: { type: "STRING", description: "Thai phonetic reading guide e.g. อะไลฟ์" },
                    pos: { type: "STRING", description: "Part of speech e.g. noun, verb, adjective" },
                    grade: { type: "STRING", description: "Suitability level: P1, P2, P3, P4, P5, P6 or Advanced" },
                    cat: { type: "STRING", description: "One of: animals, body, food, school, verbs, general" },
                    emoji: { type: "STRING", description: "Single relevant emoji icon" },
                    sentence: { type: "STRING", description: "Simple example sentence in English" },
                    sentenceTh: { type: "STRING", description: "Thai translation of the example sentence" },
                    notes: { type: "STRING", description: "Short memory tip or idiom explanation in Thai" }
                },
                required: ["word", "thai", "phonetic", "pos", "grade", "cat", "emoji", "sentence", "sentenceTh"]
            };

            const payload = {
                contents: [{ parts: [{ text: promptText }] }],
                systemInstruction: { parts: [{ text: systemInstruction }] },
                generationConfig: {
                    responseMimeType: "application/json",
                    responseSchema: schema
                }
            };

            let response = await fetch(apiUrl, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            });

            if (!response.ok) {
                throw new Error(`Gemini API Error: ${response.statusText}`);
            }

            const result = await response.json();
            const jsonText = result.candidates?.[0]?.content?.parts?.[0]?.text;
            if (!jsonText) throw new Error("No response from Gemini API");

            return JSON.parse(jsonText);
        }

        async function handleDictionarySearch(e) {
            if (e) e.preventDefault();
            const query = document.getElementById('searchInput').value.trim();
            if (!query) return;

            const loadingEl = document.getElementById('searchLoading');
            loadingEl.classList.remove('hidden');

            try {
                const prompt = `Provide dictionary data for the English word/phrase "${query}" for elementary students.`;
                const wordData = await callGeminiApiForWord(prompt);
                
                wordData.id = Date.now();
                saveCustomWordToStorage(wordData);
                
                // Set grade/category to view this searched word
                selectedGrade = 'all';
                selectedCategory = 'all';
                renderGradeButtons();
                renderCategoryButtons();
                filterWords();

                // Find index of newly added word
                const idx = filteredWords.findIndex(w => w.word.toLowerCase() === wordData.word.toLowerCase());
                if (idx !== -1) currentIndex = idx;

                document.getElementById('searchInput').value = '';
                startApp('study');
            } catch (error) {
                alert("ไม่สามารถค้นหาคำศัพท์ได้ชั่วคราว: " + error.message);
            } finally {
                loadingEl.classList.add('hidden');
            }
        }

        function playSound(type) {
            if (isMuted) return;
            try {
                if (Tone.context.state !== 'running') Tone.start();
                if (type === 'correct') {
                    synth.triggerAttackRelease("C5", "8n");
                    setTimeout(() => synth.triggerAttackRelease("E5", "8n"), 100);
                    setTimeout(() => synth.triggerAttackRelease("G5", "8n"), 200);
                } else if (type === 'wrong') {
                    synth.triggerAttackRelease("G3", "8n");
                    setTimeout(() => synth.triggerAttackRelease("C3", "8n"), 150);
                } else if (type === 'click') {
                    synth.triggerAttackRelease("C4", "16n");
                }
            } catch (err) {}
        }

        function speakText(text) {
            if ('speechSynthesis' in window) {
                window.speechSynthesis.cancel();
                const utterance = new SpeechSynthesisUtterance(text);
                utterance.lang = 'en-US';
                utterance.rate = speechRate;
                utterance.pitch = 1.0;
                window.speechSynthesis.speak(utterance);
            }
        }

                const matchCat = (selectedCategory === 'all') || (item.cat === selectedCategory);
                return matchGrade && matchCat;

        }

        function nextQuizQuestion() {
            if (quizQuestionsQueue.length === 0 || currentQuestionIndex >= quizQuestionsQueue.length) {
                quizQuestionsQueue = [...filteredWords].sort(() => 0.5 - Math.random()).slice(0, 10);
                currentQuestionIndex = 0;
            }

            if (currentQuestionIndex >= quizQuestionsQueue.length) {
                showQuizResult();
                return;
            }

            currentQuizItem = quizQuestionsQueue[currentQuestionIndex];
            renderQuizInterface();
        }

        function renderQuizInterface() {
            document.getElementById('quizQuestionNum').innerText = `ข้อที่ ${currentQuestionIndex + 1} / ${quizQuestionsQueue.length}`;
            document.getElementById('quizEmoji').innerText = currentQuizItem.emoji || '❓';
            document.getElementById('quizPhonetic').innerText = `[${currentQuizItem.phonetic || ''}]`;

            const choicesContainer = document.getElementById('quizOptionsContainer');
            const spellingContainer = document.getElementById('quizSpellingContainer');

            if (quizModeType === 'choice') {
                choicesContainer.classList.remove('hidden');
                spellingContainer.classList.add('hidden');
                document.getElementById('quizWord').innerText = currentQuizItem.word;
                document.getElementById('quizPromptText').innerText = "เลือกคำแปลภาษาไทยที่ถูกต้องที่สุด:";

                // Choices Generator
                const choices = [currentQuizItem.thai];
                const otherWords = VOCABULARY.filter(w => w.thai !== currentQuizItem.thai);
                const shuffledOthers = otherWords.sort(() => 0.5 - Math.random());

                for (let i = 0; i < 3 && i < shuffledOthers.length; i++) {
                    choices.push(shuffledOthers[i].thai);
                }
                choices.sort(() => 0.5 - Math.random());

                choicesContainer.innerHTML = '';
                choices.forEach((choiceText) => {
                    const btn = document.createElement('button');
                    btn.className = 'w-full bg-white hover:bg-slate-50 border-2 border-slate-200 hover:border-brand-pink text-slate-800 font-bold p-4 rounded-2xl shadow-sm text-center transition transform active:scale-95 text-base';
                    btn.innerText = choiceText;
                    btn.onclick = () => checkChoiceAnswer(choiceText, btn);
                    choicesContainer.appendChild(btn);
                });
            } else {
                // Spelling Mode
                choicesContainer.classList.add('hidden');
                spellingContainer.classList.remove('hidden');
                document.getElementById('quizWord').innerText = `"${currentQuizItem.thai}"`;
                document.getElementById('quizPromptText').innerText = "ฟังเสียงแล้วเขียนสะกดคำศัพท์ภาษาอังกฤษให้ถูกต้อง:";
                document.getElementById('spellingInput').value = '';
                document.getElementById('spellingInput').focus();
            }

            speakText(currentQuizItem.word);
        }

        function checkChoiceAnswer(selectedChoice, buttonElem) {
            const buttons = document.querySelectorAll('#quizOptionsContainer button');
            buttons.forEach(b => b.disabled = true);

            if (selectedChoice === currentQuizItem.thai) {
                playSound('correct');
                buttonElem.classList.remove('bg-white', 'border-slate-200');
                buttonElem.classList.add('bg-emerald-500', 'text-white', 'border-emerald-500');
                
                quizScore += 1;
                userTotalScore += 10;
                document.getElementById('quizScoreText').innerText = quizScore;
                document.getElementById('globalScoreDisplay').innerText = userTotalScore;
            } else {
                playSound('wrong');
                buttonElem.classList.remove('bg-white', 'border-slate-200');
                buttonElem.classList.add('bg-rose-500', 'text-white', 'border-rose-500');

                buttons.forEach(b => {
                    if (b.innerText === currentQuizItem.thai) {
                        b.classList.remove('bg-white', 'border-slate-200');
                        b.classList.add('bg-emerald-500', 'text-white', 'border-emerald-500');
                    }
                });
            }

            setTimeout(() => {
                currentQuestionIndex++;
                if (currentQuestionIndex < quizQuestionsQueue.length) {
                    nextQuizQuestion();
                } else {
                    showQuizResult();
                }
            }, 1400);
        }

        function handleSpellingSubmit(e) {
            e.preventDefault();
            const inputVal = document.getElementById('spellingInput').value.trim().toLowerCase();
            const correctVal = currentQuizItem.word.trim().toLowerCase();

            if (inputVal === correctVal) {
                playSound('correct');
                quizScore += 1;
                userTotalScore += 10;
                document.getElementById('quizScoreText').innerText = quizScore;
                document.getElementById('globalScoreDisplay').innerText = userTotalScore;
            } else {
                playSound('wrong');
                alert(`คำตอบไม่ถูกต้อง! คำศัพท์ที่ถูกต้องคือ "${currentQuizItem.word}"`);
            }

            currentQuestionIndex++;
            if (currentQuestionIndex < quizQuestionsQueue.length) {
                nextQuizQuestion();
            } else {
                showQuizResult();
            }
        }

        function showQuizResult() {
            hideAllViews();
            document.getElementById('resultView').classList.remove('hidden');
            
            const total = quizQuestionsQueue.length;
            document.getElementById('finalScoreDisplay').innerText = `${quizScore} / ${total}`;

            const feedback = document.getElementById('finalFeedbackText');
            if (quizScore === total) {
                feedback.innerText = "🌟 เพอร์เฟกต์มาก! ได้คะแนนเต็ม 10/10 เก่งที่สุดเลย!";
            } else if (quizScore >= total * 0.7) {
                feedback.innerText = "👏 ทำได้ดีมาก! มีความรู้ความเข้าใจคำศัพท์แน่นมาก";
            } else {
                feedback.innerText = "💪 พยายามได้ดีแล้ว! ลองกลับไปทบทวนในโหมดเรียนรู้อีกครั้งนะ";
            }
        }
